	public static void poupançaMenu() {
		
		JPanel Painel = new JPanel();
		JFrame login = new JFrame("BCraftOS - Poupança");
		login.setSize(520, 570);
		login.add(Painel);
		login.setDefaultCloseOperation(JFrame.DISPOSE_ON_CLOSE);
		login.setAlwaysOnTop(true);
		login.setResizable(false);
		login.setLayout(null);
		
		// info 
		
		JLabel poupançaTotal = new JLabel();
		poupançaTotal.setBounds(20, 100, 300, 30);
		login.add(poupançaTotal);
		
		File documentos = javax.swing.filechooser.FileSystemView
				.getFileSystemView()
				.getDefaultDirectory();
		
		File pasta = new File(documentos, "BCraftOStats");
		File pastaUsuario = new File(pasta, perfil.getNome());
		
		File poupançaMoney = new File(
					
				pastaUsuario, "Poupança.txt"
				
				);
		
		if(!poupançaMoney.exists()) {
			try {
				
				poupançaMoney.createNewFile();
				
				BufferedWriter escritor = new BufferedWriter(new FileWriter(poupançaMoney));
				escritor.write("0");
				escritor.close();
				
			} catch (IOException erro) {
				
				erro.printStackTrace();
				
			}
		}
		
		// atualiza a merda do texto
		
		javax.swing.Timer timer = new javax.swing.Timer(500, e -> {
			
			if (poupançaMoney.exists()) {
				
				try {
					
					BufferedReader leitor = new BufferedReader(new FileReader(poupançaMoney));
					String dinheiroSalvo = leitor.readLine();
					leitor.close();
					double valor = Double.parseDouble(dinheiroSalvo);
					poupançaTotal.setText(String.format("🏦 Poupança: R$ %.2f", valor));
				} catch (IOException erro) {
					
					erro.printStackTrace();
					
				}
			} else {
				poupançaTotal.setText("🏦 Poupança: R$ 0,00");
			}
			
		});
		
		timer.start();
		
		// parte que pode fazer continuação
		
		File correnteMoney = new File(
		        pastaUsuario,
		        "ContaCorrenteMoney.txt"
		);

		File poupancaMoney = new File(
		        pastaUsuario,
		        "Poupanca.txt"
		);

		// garante que a conta corrente existe
		if (!correnteMoney.exists()) {
		    try {
		        BufferedWriter escritor = new BufferedWriter(new FileWriter(correnteMoney));
		        escritor.write("0");
		        escritor.close();
		    } catch (IOException erro) {
		        erro.printStackTrace();
		    }
		}

		// garante que a poupança existe
		if (!poupancaMoney.exists()) {
		    try {
		        BufferedWriter escritor = new BufferedWriter(new FileWriter(poupancaMoney));
		        escritor.write("0");
		        escritor.close();
		    } catch (IOException erro) {
		        erro.printStackTrace();
		    }
		}

		// pergunta o valor
		JLabel perguntaValor = new JLabel("Quanto quer colocar na poupança?");
		perguntaValor.setBounds(20, 20, 300, 30);
		login.add(perguntaValor);

		// campo do valor
		JTextField valorAdicionar = new JTextField();
		valorAdicionar.setBounds(20, 150, 200, 30);
		login.add(valorAdicionar);

		// botão
		JButton adicionar = new JButton("Adicionar");
		adicionar.setBounds(230, 150, 120, 30);
		login.add(adicionar);

		// ação do botão
		adicionar.addActionListener(e -> {
		    try {
		        double valor = Double.parseDouble(valorAdicionar.getText());

		        // verifica se o valor é válido
		        if (valor <= 0) {
		            JOptionPane.showMessageDialog(
		                    login,
		                    "Digite um valor maior que zero!"
		            );
		            return;
		        }

		        // lê o saldo da conta corrente
		        BufferedReader leitorCorrente = new BufferedReader(new FileReader(correnteMoney));
		        double saldoCorrente = Double.parseDouble(leitorCorrente.readLine());
		        leitorCorrente.close();

		        // lê o saldo da poupança
		        BufferedReader leitorPoupanca = new BufferedReader(new FileReader(poupancaMoney));
		        double saldoPoupanca = Double.parseDouble(leitorPoupanca.readLine());
		        leitorPoupanca.close();

		        // verifica se tem dinheiro suficiente na corrente
		        if (valor > saldoCorrente) {
		            JOptionPane.showMessageDialog(
		                    login,
		                    "Você não tem dinheiro suficiente na conta corrente!"
		            );
		            return;
		        }

		        // faz a subtração da conta corrente
		        saldoCorrente = saldoCorrente - valor;

		        // pega o valor que foi retirado da corrente
		        double valorRetirado = valor;

		        // adiciona o valor retirado na poupança
		        saldoPoupanca = saldoPoupanca + valorRetirado;

		        // salva a nova conta corrente
		        BufferedWriter escritorCorrente = new BufferedWriter(new FileWriter(correnteMoney));
		        escritorCorrente.write(String.valueOf(saldoCorrente));
		        escritorCorrente.close();

		        // salva a nova poupança
		        BufferedWriter escritorPoupanca = new BufferedWriter(new FileWriter(poupancaMoney));
		        escritorPoupanca.write(String.valueOf(saldoPoupanca));
		        escritorPoupanca.close();

		        // limpa o campo
		        valorAdicionar.setText("");

		    } catch (NumberFormatException erro) {
		        JOptionPane.showMessageDialog(
		                login,
		                "Digite um número válido!"
		        );
		    } catch (IOException erro) {
		        erro.printStackTrace();
		    }
		});


		login.setVisible(true);
		
	}
